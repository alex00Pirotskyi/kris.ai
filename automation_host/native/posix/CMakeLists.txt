cmake_minimum_required(VERSION 3.20)
project(kristin_posix_native C)
add_executable(kristin_posix_watchdog watchdog.c)
add_executable(kristin_native_pty_probe pty_probe.c)
set_property(TARGET kristin_posix_watchdog PROPERTY C_STANDARD 11)
set_property(TARGET kristin_native_pty_probe PROPERTY C_STANDARD 11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kristin_posix_watchdog PRIVATE -Wall -Wextra -Werror)
  target_compile_options(kristin_native_pty_probe PRIVATE -Wall -Wextra -Werror)
endif()
if(UNIX AND NOT APPLE)
  target_link_libraries(kristin_native_pty_probe PRIVATE util)
endif()
