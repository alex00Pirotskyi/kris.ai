cmake_minimum_required(VERSION 3.20)
project(kristin_windows_native LANGUAGES CXX)
add_executable(kristin_job_supervisor job_supervisor.cpp)
add_executable(kristin_native_pty_probe conpty_probe.cpp)
target_compile_features(kristin_job_supervisor PRIVATE cxx_std_17)
target_compile_features(kristin_native_pty_probe PRIVATE cxx_std_17)
target_compile_definitions(kristin_job_supervisor PRIVATE UNICODE _UNICODE)
target_compile_definitions(kristin_native_pty_probe PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(kristin_job_supervisor PRIVATE /W4 /WX)
  target_compile_options(kristin_native_pty_probe PRIVATE /W4 /WX)
endif()
